extends "res://tests/fixtures/drive_twilight_expedition.gd"
func _resume_twilight() -> void:
	_keyboard = _argument("--input","controller")=="keyboard"
	_outcome = _argument("--outcome","collect")
	_sentinel_outcome = _argument("--sentinel","capture")
	_captures = OS.get_environment("BTA_EVIDENCE_DIR")
	_small = OS.get_cmdline_user_args().has("--small")
	_scene = load("res://scenes/main/main.tscn").instantiate() as ShelfPrototype
	_scene.settings_path = "user://"+_argument("--profile","bta055-route")+".json"
	_scene.expedition_path = _scene.settings_path+".expedition.json"
	root.add_child(_scene)
	root.size = Vector2i(800,600) if _small else Vector2i(1280,800)
	await _frames(10)
	var before: Dictionary = _scene.saves._state.duplicate(true)
	assert(_scene.journey.expedition.concluded())
	_scene.missions.acknowledgement_claimed.connect(func(_id: StringName,_ack: StringName) -> void: _ack_events+=1)
	_scene.story.choice_applied.connect(func(_id: StringName,_encounter: StringName) -> void: _choice_events+=1)
	await _activate(_control("Begin"))
	await _wait_transfer()
	assert(_scene.journey.snapshot()==before.journey)
	assert(_scene.journey.expedition.concluded())
	assert(_scene.submarine.global_position.distance_to(Vector2(before.player.position[0],before.player.position[1]))<1)
	assert(absf(_scene.submarine.resource_system.battery-before.player.battery)<0.2)
	var expected: float = _scene.revision.tree.sonar_radii[int(_scene.revision.campaign.ranks.sonar)]+(100.0 if _sentinel_outcome=="capture" else 0.0)
	assert(_scene.revision.capabilities().sonar==expected)
	assert(_scene.journey.snapshot()==before.journey)
	assert(_scene.saves.progress_snapshot()==before.progress)
	assert(_scene.revision.campaign==before.prototype.campaign)
	assert(_choice_events==0 and _ack_events==0)
	print("ACTUAL_BTA055_CONTINUE_PASS phase=",before.journey.phase," input=",("keyboard" if _keyboard else "controller")," no_replayed_events=true")
	await _finish_process()
